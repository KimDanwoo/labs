타입 좁히기는 넓은 타입을 좁은 타입으로 좁혀가는 과정이며, 타입 가드는 이를 수행하는 표현식입니다.

```typescript
// 1. typeof 가드
function process(value: string | number) {
  if (typeof value === 'string') {
    return value.toUpperCase(); // string으로 좁혀짐
  }
  return value.toFixed(2); // number로 좁혀짐
}

// 2. instanceof 가드
function handleError(error: Error | string) {
  if (error instanceof Error) {
    return error.message;
  }
  return error;
}

// 3. in 가드
interface Dog { bark(): void; }
interface Cat { meow(): void; }

function speak(animal: Dog | Cat) {
  if ('bark' in animal) {
    animal.bark();
  } else {
    animal.meow();
  }
}

// 4. 커스텀 타입 가드 (is 키워드)
function isString(value: unknown): value is string {
  return typeof value === 'string';
}

function process2(val: unknown) {
  if (isString(val)) {
    console.log(val.toUpperCase()); // string으로 좁혀짐
  }
}
```

- 커스텀 타입 가드(`is` 키워드)는 복잡한 타입 판별에 유용합니다.
- 판별된 유니온(Discriminated Union)도 타입 좁히기에 자주 사용됩니다.
