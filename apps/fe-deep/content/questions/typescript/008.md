공통 리터럴 속성(판별자)으로 유니온의 각 케이스를 구분하는 패턴입니다.

```typescript
// API 응답 타입
type ApiResult<T> =
  | { status: 'success'; data: T }
  | { status: 'error'; error: string; code: number }
  | { status: 'loading' };

function handleResult(result: ApiResult<User>) {
  switch (result.status) {
    case 'success':
      console.log(result.data); // data 타입 자동 추론
      break;
    case 'error':
      console.log(result.error, result.code); // error, code 접근 가능
      break;
    case 'loading':
      showSpinner();
      break;
  }
}

// 이벤트 시스템
type AppEvent =
  | { type: 'CLICK'; x: number; y: number }
  | { type: 'KEY_PRESS'; key: string }
  | { type: 'SCROLL'; offset: number };
```

**장점:**
- switch문에서 각 케이스의 타입이 자동으로 좁혀짐(narrowing)
- exhaustiveness check로 모든 케이스 처리 보장
- Redux 액션, 이벤트 시스템, 상태 머신에 적합

```typescript
// exhaustiveness check
function assertNever(x: never): never { throw new Error('Unexpected'); }
switch (event.type) {
  case 'CLICK': break;
  case 'KEY_PRESS': break;
  default: assertNever(event); // SCROLL 누락 시 컴파일 에러
}
```
